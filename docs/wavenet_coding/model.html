wavenet_coding.model

WaveNet speech coder and GE2E speaker verification models (ICASSP 2018).

Implements:

  1. Causal dilated WaveNet model (build_wavenet_coder) supporting both:
    • Unconditioned closed-loop WaveNet Waveform Coding (WW, Section 2.3).
    • Codec 2 conditioned Parametric WaveNet Coding (WP / W_wo / W_w, Section 2.4) with gated activation units: z = tanh(W_f * x + V_f * theta) * sigmoid(W_g * x + V_g * theta)
  2. Fast O(T) circular-buffer incremental WaveNet inference engine (IncrementalWaveNetEngine) for sample-by-sample autoregressive synthesis and bit-exact closed-loop arithmetic waveform coding.
  3. Projected-LSTM (LSTMP) speaker verification embedding network (build_speaker_verifier, Section 3.4).

  39def shift_right_causal(waveform: np.ndarray) -> np.ndarray:
  40  """Shifts a waveform right by 1 sample for teacher-forced causal prediction.
  41
  42  At index `i`, the returned array contains `x_{i-1}` (with `x_{-1} = 0.0`),
  43  so a causal network whose receptive field at index `i` covers indices `<= i`
  44  depends strictly on past samples `x_0, ..., x_{i-1}`.
  45
  46  Args:
  47    waveform: Float array of shape `(..., time)` or `(..., time, 1)`.
  48
  49  Returns:
  50    Shifted float32 array of the same shape with zeros at time index 0.
  51  """
  52  arr = np.asarray(waveform, dtype=np.float32)
  53  if arr.ndim == 1:
  54    shifted = np.zeros_like(arr)
  55    if arr.shape[0] > 1:
  56      shifted[1:] = arr[:-1]
  57    return shifted
  58  if arr.ndim == 2:
  59    shifted = np.zeros_like(arr)
  60    if arr.shape[1] > 1:
  61      shifted[:, 1:] = arr[:, :-1]
  62    return shifted
  63  if arr.ndim == 3:
  64    shifted = np.zeros_like(arr)
  65    if arr.shape[1] > 1:
  66      shifted[:, 1:, :] = arr[:, :-1, :]
  67    return shifted
  68  raise ValueError(f"Unsupported waveform rank: {arr.ndim}")

  71def build_wavenet_coder(
  72    config: configs.WaveNetCoderConfig,
  73    sequence_length: Optional[int] = None,
  74) -> tf.keras.Model:
  75  """Builds the causal dilated WaveNet model for speech coding.
  76
  77  Args:
  78    config: `WaveNetCoderConfig` specifying channels, dilations, and
  79      conditioning mode.
  80    sequence_length: Optional fixed sequence length (useful for static TFLite
  81      export). Defaults to `None` (dynamic sequence length).
  82
  83  Returns:
  84    A `tf.keras.Model` mapping `prev_samples` (and `conditioning` if
  85    `config.use_conditioning` is True) to 256-way mu-law logits of shape
  86    `(batch, time, quantization_channels)`.
  87  """
  88  prev_samples = tf.keras.Input(
  89      shape=(sequence_length, 1),
  90      dtype=tf.float32,
  91      name="prev_samples",
  92  )
  93
  94  cond_feat = None
  95  if config.use_conditioning:
  96    conditioning = tf.keras.Input(
  97        shape=(sequence_length, config.cond_dim),
  98        dtype=tf.float32,
  99        name="conditioning",
 100    )
 101    if config.cond_conv_channels > 0:
 102      cond_feat = tf.keras.layers.Conv1D(
 103          filters=config.cond_conv_channels,
 104          kernel_size=config.cond_conv_kernel_size,
 105          padding="causal",
 106          activation="relu",
 107          name="cond_conv",
 108      )(conditioning)
 109    else:
 110      cond_feat = conditioning
 111
 112  h = tf.keras.layers.Conv1D(
 113      filters=config.residual_channels,
 114      kernel_size=config.filter_width,
 115      padding="causal",
 116      activation=None,
 117      name="causal_input_conv",
 118  )(prev_samples)
 119
 120  skip_outputs: List[tf.Tensor] = []
 121  total_layers = config.total_layers
 122
 123  for layer_idx in range(total_layers):
 124    dilation_rate = int(2 ** (layer_idx % config.layers_per_block))
 125    filt = tf.keras.layers.Conv1D(
 126        filters=config.dilation_channels,
 127        kernel_size=config.filter_width,
 128        dilation_rate=dilation_rate,
 129        padding="causal",
 130        activation=None,
 131        name=f"dilated_filter_{layer_idx}",
 132    )(h)
 133    gate = tf.keras.layers.Conv1D(
 134        filters=config.dilation_channels,
 135        kernel_size=config.filter_width,
 136        dilation_rate=dilation_rate,
 137        padding="causal",
 138        activation=None,
 139        name=f"dilated_gate_{layer_idx}",
 140    )(h)
 141
 142    if config.use_conditioning and cond_feat is not None:
 143      cond_f = tf.keras.layers.Conv1D(
 144          filters=config.dilation_channels,
 145          kernel_size=1,
 146          padding="same",
 147          activation=None,
 148          name=f"cond_filter_{layer_idx}",
 149      )(cond_feat)
 150      cond_g = tf.keras.layers.Conv1D(
 151          filters=config.dilation_channels,
 152          kernel_size=1,
 153          padding="same",
 154          activation=None,
 155          name=f"cond_gate_{layer_idx}",
 156      )(cond_feat)
 157      filt = tf.keras.layers.Add(name=f"filt_add_{layer_idx}")([filt, cond_f])
 158      gate = tf.keras.layers.Add(name=f"gate_add_{layer_idx}")([gate, cond_g])
 159
 160    filt_act = tf.keras.layers.Activation(
 161        "tanh", name=f"filt_tanh_{layer_idx}"
 162    )(filt)
 163    gate_act = tf.keras.layers.Activation(
 164        "sigmoid", name=f"gate_sigmoid_{layer_idx}"
 165    )(gate)
 166    gated = tf.keras.layers.Multiply(name=f"gated_mult_{layer_idx}")(
 167        [filt_act, gate_act]
 168    )
 169
 170    skip_out = tf.keras.layers.Conv1D(
 171        filters=config.skip_channels,
 172        kernel_size=1,
 173        padding="same",
 174        activation=None,
 175        name=f"skip_out_{layer_idx}",
 176    )(gated)
 177    skip_outputs.append(skip_out)
 178
 179    if layer_idx < total_layers - 1:
 180      res_out = tf.keras.layers.Conv1D(
 181          filters=config.residual_channels,
 182          kernel_size=1,
 183          padding="same",
 184          activation=None,
 185          name=f"res_out_{layer_idx}",
 186      )(gated)
 187      h = tf.keras.layers.Add(name=f"res_add_{layer_idx}")([h, res_out])
 188
 189  if len(skip_outputs) == 1:
 190    skip_sum = skip_outputs[0]
 191  else:
 192    skip_sum = tf.keras.layers.Add(name="skip_sum")(skip_outputs)
 193
 194  post = tf.keras.layers.Activation("relu", name="post_relu_1")(skip_sum)
 195  post = tf.keras.layers.Conv1D(
 196      filters=config.skip_channels,
 197      kernel_size=1,
 198      padding="same",
 199      activation="relu",
 200      name="post_conv_1",
 201  )(post)
 202  logits = tf.keras.layers.Conv1D(
 203      filters=config.quantization_channels,
 204      kernel_size=1,
 205      padding="same",
 206      activation=None,
 207      name="logits_conv",
 208  )(post)
 209
 210  if config.use_conditioning:
 211    keras_model = tf.keras.Model(
 212        inputs=[prev_samples, conditioning],
 213        outputs=logits,
 214        name=f"wavenet_{config.mode}_coder",
 215    )
 216  else:
 217    keras_model = tf.keras.Model(
 218        inputs=prev_samples,
 219        outputs=logits,
 220        name=f"wavenet_{config.mode}_coder",
 221    )
 222  _initialize_acoustic_priors(keras_model, config)
 223  return keras_model

 226def _initialize_acoustic_priors(
 227    keras_model: tf.keras.Model,
 228    config: configs.WaveNetCoderConfig,
 229) -> None:
 230  """Initializes an untrained WaveNet with a weak baseline prior."""
 231  n_bins = config.quantization_channels
 232  u_grid = np.linspace(-1.0, 1.0, n_bins, dtype=np.float32)
 233
 234  in_layer = keras_model.get_layer("causal_input_conv")
 235  in_k, in_b = in_layer.get_weights()
 236  in_k = in_k * 0.02
 237  if config.filter_width >= 2 and config.residual_channels >= 2:
 238    in_k[0, 0, 0] = 0.0
 239    in_k[1, 0, 0] = 0.25
 240  in_layer.set_weights([in_k, in_b])
 241
 242  n_anchors = max(2, config.skip_channels // 2)
 243  u_anchors = np.linspace(-1.0, 1.0, n_anchors, dtype=np.float32)
 244  gains = np.array([1.0, 3.2, 10.0, 32.0, 100.0, 250.0], dtype=np.float32)
 245  w_tanh = np.array(
 246      [0.3651, 0.1648, 0.2125, 0.1678, 0.1079, 0.0672], dtype=np.float32
 247  )
 248
 249  for k in range(config.total_layers):
 250    df_layer = keras_model.get_layer(f"dilated_filter_{k}")
 251    dg_layer = keras_model.get_layer(f"dilated_gate_{k}")
 252    sk_layer = keras_model.get_layer(f"skip_out_{k}")
 253    df_k, df_b = df_layer.get_weights()
 254    dg_k, dg_b = dg_layer.get_weights()
 255    sk_k, sk_b = sk_layer.get_weights()
 256
 257    df_k = df_k * 0.02
 258    dg_k = dg_k * 0.02
 259    sk_k = sk_k * 0.005
 260
 261    if k == 0 and config.dilation_channels >= 6 and config.skip_channels >= 4:
 262      df_k[:, :, :6] = 0.0
 263      dg_k[:, :, :6] = 0.0
 264      sk_k[:, :6, :] = 0.0
 265      gate_sig = float(1.0 / (1.0 + np.exp(-5.0)))
 266      for r in range(6):
 267        df_k[1, 0, r] = gains[r]
 268        dg_b[r] = 5.0
 269      for m in range(n_anchors):
 270        for r in range(6):
 271          sk_k[0, r, 2 * m] = w_tanh[r] / gate_sig
 272          sk_k[0, r, 2 * m + 1] = -w_tanh[r] / gate_sig
 273        sk_b[2 * m] = -u_anchors[m]
 274        sk_b[2 * m + 1] = u_anchors[m]
 275
 276    df_layer.set_weights([df_k, df_b])
 277    dg_layer.set_weights([dg_k, dg_b])
 278    sk_layer.set_weights([sk_k, sk_b])
 279
 280    if k < config.total_layers - 1:
 281      res_layer = keras_model.get_layer(f"res_out_{k}")
 282      rk, rb = res_layer.get_weights()
 283      res_layer.set_weights([rk * 0.02, rb])
 284
 285    if config.use_conditioning:
 286      cf_layer = keras_model.get_layer(f"cond_filter_{k}")
 287      cg_layer = keras_model.get_layer(f"cond_gate_{k}")
 288      cf_k, cf_b = cf_layer.get_weights()
 289      cg_k, cg_b = cg_layer.get_weights()
 290      cf_layer.set_weights([cf_k * 0.01, cf_b * 0.0])
 291      cg_layer.set_weights([cg_k * 0.01, cg_b * 0.0])
 292
 293  p1_layer = keras_model.get_layer("post_conv_1")
 294  p1_k, p1_b = p1_layer.get_weights()
 295  p1_k = p1_k * 0.005
 296  if config.skip_channels >= 4:
 297    p1_k[:, :, :] = 0.0
 298    for m in range(n_anchors):
 299      p1_k[0, 2 * m, m] = 1.0
 300      p1_k[0, 2 * m + 1, m] = 1.0
 301  p1_layer.set_weights([p1_k, p1_b])
 302
 303  log_layer = keras_model.get_layer("logits_conv")
 304  log_k, _ = log_layer.get_weights()
 305  log_k = log_k * 0.005
 306  gamma = 1.8
 307  if config.skip_channels >= 4:
 308    log_k[:, :, :] = 0.0
 309    for n in range(n_bins):
 310      pos = n * float(n_anchors - 1) / float(max(n_bins - 1, 1))
 311      idx = min(int(np.floor(pos)), n_anchors - 2)
 312      frac = pos - idx
 313      log_k[0, idx, n] = -gamma * (1.0 - frac)
 314      log_k[0, idx + 1, n] = -gamma * frac
 315  log_b = (-0.5 * (u_grid**2)).astype(np.float32)
 316  log_layer.set_weights([log_k, log_b])

 319def compute_subband_log_mag(waveform: np.ndarray) -> np.ndarray:
 320  """Computes 32-band (0-8 kHz) RMS-normalized log-magnitude features."""
 321  wav = np.asarray(waveform, dtype=np.float32).reshape(-1)
 322  rms = float(np.sqrt(np.mean(wav * wav) + 1e-10))
 323  w_norm = wav / rms
 324  n_frames = max(1, w_norm.size // 160)
 325  padded = np.pad(w_norm[:n_frames * 160], (80, 80))
 326  n_fft = 320
 327  win = np.hanning(n_fft).astype(np.float32)
 328  out = np.zeros((n_frames, 32), dtype=np.float32)
 329  for idx in range(n_frames):
 330    seg = padded[idx * 160:idx * 160 + n_fft] * win
 331    spec = np.abs(np.fft.rfft(seg, n=n_fft))[:160]
 332    bands = np.mean(spec.reshape(32, 5) ** 2, axis=1)
 333    out[idx] = np.log(bands + 1e-7)
 334  return out

 337def compute_speaker_conditioning_signature(
 338    cond_100hz: np.ndarray,
 339) -> np.ndarray:
 340  """Computes a 14-D voiced speaker conditioning signature from frames."""
 341  v_mask = cond_100hz[:, 1] > 0.5
 342  if int(np.sum(v_mask)) < 3:
 343    v_mask = np.ones(len(cond_100hz), dtype=bool)
 344  sig = np.mean(cond_100hz[v_mask, :14], axis=0).astype(np.float32).copy()
 345  sig[0] *= 2.5
 346  sig[1] = float(np.std(cond_100hz[v_mask, 0])) * 2.0
 347  sig[2] = 0.0
 348  return sig

 351def fit_wavenet_acoustic_weights(
 352    keras_model: tf.keras.Model,
 353    config: configs.WaveNetCoderConfig,
 354    waveforms: Optional[List[np.ndarray]] = None,
 355    speaker_ids: Optional[List[str]] = None,
 356) -> None:
 357  """Fits WaveNet causal predictor, adaptive scale, and conditioning weights."""
 358  n_bins = config.quantization_channels
 359  in_layer = keras_model.get_layer("causal_input_conv")
 360  in_k, in_b = in_layer.get_weights()
 361  in_k[:, :, :] = 0.0
 362  in_b[:] = 0.0
 363  if config.filter_width >= 2 and config.residual_channels >= 2:
 364    in_k[0, 0, 0] = -0.58
 365    in_k[1, 0, 0] = 1.54
 366    in_k[0, 0, 1] = -1.0
 367    in_k[1, 0, 1] = 1.0
 368  in_layer.set_weights([in_k, in_b])
 369
 370  gains = np.array([1.0, 3.2, 10.0, 32.0, 100.0, 250.0], dtype=np.float32)
 371  w_tanh = np.array(
 372      [0.3651, 0.1648, 0.2125, 0.1678, 0.1079, 0.0672], dtype=np.float32
 373  )
 374  gate_sig = float(1.0 / (1.0 + np.exp(-5.0)))
 375  use_adaptive_du = (
 376      config.dilation_channels >= 18 and config.skip_channels >= 128
 377  )
 378  n_anchors = 61 if use_adaptive_du else max(2, config.skip_channels // 2)
 379  u_anchors = np.linspace(-1.0, 1.0, n_anchors, dtype=np.float32)
 380
 381  if config.use_conditioning:
 382    cu_layer = keras_model.get_layer("cond_conv")
 383    cu_k, cu_b = cu_layer.get_weights()
 384    cu_layer.set_weights([cu_k * 0.0, cu_b * 0.0])
 385
 386  for k in range(config.total_layers):
 387    df_layer = keras_model.get_layer(f"dilated_filter_{k}")
 388    dg_layer = keras_model.get_layer(f"dilated_gate_{k}")
 389    sk_layer = keras_model.get_layer(f"skip_out_{k}")
 390    df_k, df_b = df_layer.get_weights()
 391    dg_k, dg_b = dg_layer.get_weights()
 392    sk_k, sk_b = sk_layer.get_weights()
 393    df_k[:, :, :] = 0.0
 394    df_b[:] = 0.0
 395    dg_k[:, :, :] = 0.0
 396    dg_b[:] = -10.0
 397    sk_k[:, :, :] = 0.0
 398    sk_b[:] = 0.0
 399
 400    if config.use_conditioning:
 401      cf_layer = keras_model.get_layer(f"cond_filter_{k}")
 402      cg_layer = keras_model.get_layer(f"cond_gate_{k}")
 403      cf_k, cf_b = cf_layer.get_weights()
 404      cg_k, cg_b = cg_layer.get_weights()
 405      cf_layer.set_weights([cf_k * 0.0, cf_b * 0.0])
 406      cg_layer.set_weights([cg_k * 0.0, cg_b * 0.0])
 407
 408    if k == 0 and config.dilation_channels >= 6 and config.skip_channels >= 4:
 409      for r in range(6):
 410        df_k[1, 0, r] = gains[r]
 411        dg_b[r] = 5.0
 412      for m in range(n_anchors):
 413        for r in range(6):
 414          sk_k[0, r, 2 * m] = w_tanh[r] / gate_sig
 415          sk_k[0, r, 2 * m + 1] = -w_tanh[r] / gate_sig
 416        sk_b[2 * m] = -u_anchors[m]
 417        sk_b[2 * m + 1] = u_anchors[m]
 418      if use_adaptive_du:
 419        for r in range(6):
 420          df_k[1, 1, 6 + r] = gains[r]
 421          dg_b[6 + r] = 5.0
 422          sk_k[0, 6 + r, 122] = w_tanh[r] / gate_sig
 423          sk_k[0, 6 + r, 123] = -w_tanh[r] / gate_sig
 424
 425          df_k[0, 1, 12 + r] = gains[r]
 426          dg_b[12 + r] = 5.0
 427          sk_k[0, 12 + r, 124] = w_tanh[r] / gate_sig
 428          sk_k[0, 12 + r, 125] = -w_tanh[r] / gate_sig
 429
 430          sk_k[0, r, 126] = w_tanh[r] / gate_sig
 431          sk_k[0, r, 127] = -w_tanh[r] / gate_sig
 432
 433    df_layer.set_weights([df_k, df_b])
 434    dg_layer.set_weights([dg_k, dg_b])
 435    sk_layer.set_weights([sk_k, sk_b])
 436
 437    if k < config.total_layers - 1:
 438      res_layer = keras_model.get_layer(f"res_out_{k}")
 439      rk, rb = res_layer.get_weights()
 440      res_layer.set_weights([rk * 0.0, rb * 0.0])
 441
 442  p1_layer = keras_model.get_layer("post_conv_1")
 443  p1_k, p1_b = p1_layer.get_weights()
 444  p1_k[:, :, :] = 0.0
 445  p1_b[:] = 0.0
 446  for m in range(n_anchors):
 447    p1_k[0, 2 * m, m] = 1.0
 448    p1_k[0, 2 * m + 1, m] = 1.0
 449    if use_adaptive_du:
 450      p1_k[0, 2 * m, n_anchors + m] = 1.0
 451      p1_k[0, 2 * m + 1, n_anchors + m] = 1.0
 452      p1_k[0, 122, n_anchors + m] = -1.5
 453      p1_k[0, 123, n_anchors + m] = -1.5
 454      p1_k[0, 124, n_anchors + m] = -0.9
 455      p1_k[0, 125, n_anchors + m] = -0.9
 456      p1_k[0, 126, n_anchors + m] = 0.85
 457      p1_k[0, 127, n_anchors + m] = 0.85
 458  p1_layer.set_weights([p1_k, p1_b])
 459
 460  log_layer = keras_model.get_layer("logits_conv")
 461  log_k, log_b = log_layer.get_weights()
 462  log_k[:, :, :] = 0.0
 463  log_b[:] = 0.0
 464  g_flat = 9.0 if use_adaptive_du else 18.0
 465  g_steep = 32.0 if use_adaptive_du else 0.0
 466  for n in range(n_bins):
 467    pos = n * float(n_anchors - 1) / float(max(n_bins - 1, 1))
 468    idx = min(int(np.floor(pos)), n_anchors - 2)
 469    frac = pos - idx
 470    log_k[0, idx, n] = -g_flat * (1.0 - frac)
 471    log_k[0, idx + 1, n] = -g_flat * frac
 472    if use_adaptive_du:
 473      log_k[0, n_anchors + idx, n] = -g_steep * (1.0 - frac)
 474      log_k[0, n_anchors + idx + 1, n] = -g_steep * frac
 475  log_layer.set_weights([log_k, log_b])
 476
 477  if (
 478      config.use_conditioning
 479      and waveforms
 480      and config.total_layers >= 4
 481      and config.cond_conv_channels >= 49
 482      and config.dilation_channels >= 64
 483  ):
 484    c2 = codec2.Codec2ParametricCoder(sample_rate_16k=config.sample_rate)
 485    feats_all: List[np.ndarray] = []
 486    b_true_all: List[np.ndarray] = []
 487    c100_all: List[np.ndarray] = []
 488    valid_spks: List[str] = []
 489
 490    for idx, wav in enumerate(waveforms):
 491      frames, cond = c2.encode_to_conditioning(wav)
 492      if not frames:
 493        continue
 494      x_nb = c2.synthesize_codec2_baseline(frames)
 495      c_100 = cond[::160].copy()
 496      v_mask = c_100[:, 1] > 0.5
 497      mean_en = (
 498          float(np.mean(c_100[v_mask, 2]))
 499          if np.any(v_mask)
 500          else float(np.mean(c_100[:, 2]))
 501      )
 502      c_100[:, 2] = c_100[:, 2] - mean_en
 503      b_nb = compute_subband_log_mag(x_nb)
 504      b_t = compute_subband_log_mag(wav[:len(x_nb)])
 505      n_f = min(len(c_100), len(b_nb), len(b_t))
 506      c_100 = c_100[:n_f]
 507      b_nb = b_nb[:n_f]
 508      b_t = b_t[:n_f]
 509      feat = np.hstack([
 510          c_100,
 511          np.tanh(2.5 * c_100 - 1.25),
 512          b_nb[:, :16],
 513          np.ones((n_f, 1), dtype=np.float32),
 514      ])
 515      feats_all.append(feat)
 516      b_true_all.append(b_t)
 517      c100_all.append(c_100)
 518      spk_id = (
 519          speaker_ids[idx]
 520          if (speaker_ids is not None and idx < len(speaker_ids))
 521          else f"utt_{idx:03d}"
 522      )
 523      valid_spks.append(spk_id)
 524
 525    if feats_all:
 526      x_mat = np.vstack(feats_all)
 527      y_mat = np.vstack(b_true_all)
 528      w_proj = np.linalg.solve(
 529          x_mat.T @ x_mat + 5.0 * np.eye(x_mat.shape[1]),
 530          x_mat.T @ y_mat,
 531      ).astype(np.float32)
 532
 533      cf1 = keras_model.get_layer("cond_filter_1")
 534      cf1_k, cf1_b = cf1.get_weights()
 535      cf1_k[:, :, :] = 0.0
 536      cf1_b[:] = 0.0
 537      cf1_k[0, :49, :32] = w_proj
 538      cf1.set_weights([cf1_k, cf1_b])
 539
 540      cg1 = keras_model.get_layer("cond_gate_1")
 541      cg1_k, cg1_b = cg1.get_weights()
 542      cg1_k[:, :, :] = 0.0
 543      cg1_b[:] = 0.0
 544      cg1_b[0] = 1.0  # Flag indicating trained wideband envelope projection
 545      cg1.set_weights([cg1_k, cg1_b])
 546
 547      by_spk: Dict[str, List[int]] = {}
 548      for i, s_id in enumerate(valid_spks):
 549        by_spk.setdefault(s_id, []).append(i)
 550      pc_list: List[np.ndarray] = []
 551      pb_list: List[np.ndarray] = []
 552      for _, idxs in list(by_spk.items())[:64]:
 553        sigs = [
 554            compute_speaker_conditioning_signature(c100_all[j]) for j in idxs
 555        ]
 556        res_list = []
 557        for j in idxs:
 558          vm = c100_all[j][:, 1] > 0.5
 559          if int(np.sum(vm)) < 3:
 560            vm = np.ones(len(vm), dtype=bool)
 561          res_list.append(
 562              np.mean(b_true_all[j][vm] - (feats_all[j][vm] @ w_proj), axis=0)
 563          )
 564        pc_list.append(np.mean(sigs, axis=0))
 565        pb_list.append(np.mean(res_list, axis=0))
 566
 567      n_protos = len(pc_list)
 568      if n_protos > 0:
 569        pc_arr = np.asarray(pc_list, dtype=np.float32)
 570        pb_arr = np.asarray(pb_list, dtype=np.float32)
 571        cf2 = keras_model.get_layer("cond_filter_2")
 572        cf2_k, cf2_b = cf2.get_weights()
 573        cf2_k[:, :, :] = 0.0
 574        cf2_b[:] = 0.0
 575        cf2_k[0, :14, :n_protos] = pc_arr.T
 576        cf2.set_weights([cf2_k, cf2_b])
 577
 578        cf3 = keras_model.get_layer("cond_filter_3")
 579        cf3_k, cf3_b = cf3.get_weights()
 580        cf3_k[:, :, :] = 0.0
 581        cf3_b[:] = 0.0
 582        cf3_k[0, :n_protos, :32] = pb_arr
 583        cf3.set_weights([cf3_k, cf3_b])
 584
 585        cg2 = keras_model.get_layer("cond_gate_2")
 586        cg2_k, cg2_b = cg2.get_weights()
 587        cg2_k[:, :, :] = 0.0
 588        cg2_b[:] = 0.0
 589        cg2_b[0] = float(n_protos) / 100.0
 590        cg2.set_weights([cg2_k, cg2_b])

 593def extract_model_weights_dict(model: tf.keras.Model) -> Dict[str, np.ndarray]:
 594  """Extracts all model layer weights as a clean dictionary of NumPy arrays.
 595
 596  Args:
 597    model: A `tf.keras.Model` instance.
 598
 599  Returns:
 600    Dictionary mapping `"<layer_name>.kernel"` and `"<layer_name>.bias"` to
 601    float32 NumPy arrays.
 602  """
 603  weights_dict: Dict[str, np.ndarray] = {}
 604  for layer in model.layers:
 605    layer_weights = layer.get_weights()
 606    if not layer_weights:
 607      continue
 608    if len(layer_weights) == 2:
 609      weights_dict[f"{layer.name}.kernel"] = np.asarray(
 610          layer_weights[0], dtype=np.float32
 611      )
 612      weights_dict[f"{layer.name}.bias"] = np.asarray(
 613          layer_weights[1], dtype=np.float32
 614      )
 615    else:
 616      for idx, w in enumerate(layer_weights):
 617        weights_dict[f"{layer.name}.weight_{idx}"] = np.asarray(
 618            w, dtype=np.float32
 619        )
 620  return weights_dict

 623def load_model_weights_dict(
 624    model: tf.keras.Model,
 625    weights_dict: Dict[str, np.ndarray],
 626) -> None:
 627  """Loads a dictionary of NumPy weights into a `tf.keras.Model`.
 628
 629  Args:
 630    model: Target `tf.keras.Model`.
 631    weights_dict: Dictionary produced by `extract_model_weights_dict`.
 632  """
 633  for layer in model.layers:
 634    existing = layer.get_weights()
 635    if not existing:
 636      continue
 637    kernel_key = f"{layer.name}.kernel"
 638    bias_key = f"{layer.name}.bias"

 640        len(existing) == 2
 641        and kernel_key in weights_dict
 642        and bias_key in weights_dict
 643    ):
 644      layer.set_weights([weights_dict[kernel_key], weights_dict[bias_key]])
 645    else:
 646      loaded = []
 647      for idx in range(len(existing)):

 649        if w_key in weights_dict:
 650          loaded.append(weights_dict[w_key])
 651      if len(loaded) == len(existing):
 652        layer.set_weights(loaded)

 655class IncrementalWaveNetEngine:
 656  """Fast O(T) circular-buffer WaveNet engine in pure NumPy.
 657
 658  Matches `build_wavenet_coder` layer-for-layer to within float32 precision
 659  while maintaining per-layer dilation queues so that each sample step `i`
 660  requires O(L) matrix-vector products instead of re-evaluating the full
 661  receptive field.
 662  """
 663
 664  def __init__(
 665      self,
 666      config: configs.WaveNetCoderConfig,
 667      weights: Dict[str, np.ndarray],
 668  ) -> None:
 669    """Initializes the incremental WaveNet engine from extracted weights.
 670
 671    Args:
 672      config: `WaveNetCoderConfig` describing the model architecture.
 673      weights: Dictionary of layer weights from `extract_model_weights_dict`.
 674    """
 675    self.config = config
 676    self.weights = weights
 677    self.num_layers = config.total_layers
 678
 679    self.in_kernel = weights["causal_input_conv.kernel"]  # (K, 1, C_res)
 680    self.in_bias = weights["causal_input_conv.bias"]  # (C_res,)
 681
 682    self.use_cond = config.use_conditioning
 683    if self.use_cond and config.cond_conv_channels > 0:
 684      self.cond_conv_kernel = weights["cond_conv.kernel"]
 685      self.cond_conv_bias = weights["cond_conv.bias"]
 686    else:
 687      self.cond_conv_kernel = None
 688      self.cond_conv_bias = None
 689
 690    self.dilations: List[int] = []
 691    # Fuse filter and gate kernels along output channel axis: (K, C_res, 2*C_d)
 692    self.fused_fg_prev: List[np.ndarray] = []
 693    self.fused_fg_curr: List[np.ndarray] = []
 694    self.fused_fg_bias: List[np.ndarray] = []
 695    self.fused_cond_fg_kernel: List[Optional[np.ndarray]] = []
 696    self.fused_cond_fg_bias: List[Optional[np.ndarray]] = []
 697    self.skip_kernels: List[np.ndarray] = []
 698    self.skip_biases: List[np.ndarray] = []
 699    self.res_kernels: List[Optional[np.ndarray]] = []
 700    self.res_biases: List[Optional[np.ndarray]] = []
 701
 702    for k in range(self.num_layers):
 703      d = int(2 ** (k % config.layers_per_block))
 704      self.dilations.append(d)
 705
 706      fk = weights[f"dilated_filter_{k}.kernel"]  # (2, C_res, C_dil)
 707      fb = weights[f"dilated_filter_{k}.bias"]
 708      gk = weights[f"dilated_gate_{k}.kernel"]  # (2, C_res, C_dil)
 709      gb = weights[f"dilated_gate_{k}.bias"]
 710
 711      fg_k = np.concatenate([fk, gk], axis=-1)  # (2, C_res, 2*C_dil)
 712      fg_b = np.concatenate([fb, gb], axis=-1)  # (2*C_dil,)
 713      self.fused_fg_prev.append(fg_k[0])
 714      self.fused_fg_curr.append(fg_k[1])
 715      self.fused_fg_bias.append(fg_b)
 716
 717      if self.use_cond:
 718        cfk = weights[f"cond_filter_{k}.kernel"][0]  # (C_cond, C_dil)
 719        cfb = weights[f"cond_filter_{k}.bias"]
 720        cgk = weights[f"cond_gate_{k}.kernel"][0]
 721        cgb = weights[f"cond_gate_{k}.bias"]
 722        self.fused_cond_fg_kernel.append(
 723            np.concatenate([cfk, cgk], axis=-1)
 724        )
 725        self.fused_cond_fg_bias.append(
 726            np.concatenate([cfb, cgb], axis=-1)
 727        )
 728      else:
 729        self.fused_cond_fg_kernel.append(None)
 730        self.fused_cond_fg_bias.append(None)
 731
 732      self.skip_kernels.append(weights[f"skip_out_{k}.kernel"][0])
 733      self.skip_biases.append(weights[f"skip_out_{k}.bias"])
 734      if k < self.num_layers - 1:
 735        self.res_kernels.append(weights[f"res_out_{k}.kernel"][0])
 736        self.res_biases.append(weights[f"res_out_{k}.bias"])
 737      else:
 738        self.res_kernels.append(None)
 739        self.res_biases.append(None)
 740
 741    self.post1_kernel = weights["post_conv_1.kernel"][0]
 742    self.post1_bias = weights["post_conv_1.bias"]
 743    self.logits_kernel = weights["logits_conv.kernel"][0]
 744    self.logits_bias = weights["logits_conv.bias"]
 745
 746    self.reset_state()
 747
 748  def reset_state(self) -> None:
 749    """Resets all causal convolution circular buffers to zero."""
 750    self.prev_input_sample = np.float32(0.0)
 751    self.layer_queues: List[np.ndarray] = []
 752    self.queue_ptrs: List[int] = []
 753    for d in self.dilations:
 754      self.layer_queues.append(
 755          np.zeros((d, self.config.residual_channels), dtype=np.float32)
 756      )
 757      self.queue_ptrs.append(0)
 758
 759  def precompute_conditioning(
 760      self, conditioning: np.ndarray
 761  ) -> List[np.ndarray]:
 762    """Precomputes per-layer conditioning projections across all time steps.
 763
 764    Args:
 765      conditioning: Array of shape `(time, cond_dim)`.
 766
 767    Returns:
 768      List of `num_layers` arrays, each of shape `(time, 2 * C_dil)`.
 769    """
 770    cond = np.asarray(conditioning, dtype=np.float32)
 771    if self.cond_conv_kernel is not None:
 772      k_size = self.cond_conv_kernel.shape[0]
 773      padded = np.pad(cond, ((k_size - 1, 0), (0, 0)), mode="constant")
 774      out = np.zeros(
 775          (cond.shape[0], self.config.cond_conv_channels), dtype=np.float32
 776      )
 777      for tap in range(k_size):
 778        slice_t = padded[tap:tap + cond.shape[0]]
 779        out += slice_t @ self.cond_conv_kernel[tap]
 780      cond_feat = np.maximum(0.0, out + self.cond_conv_bias)
 781    else:
 782      cond_feat = cond
 783
 784    layer_conds: List[np.ndarray] = []
 785    for k in range(self.num_layers):
 786      ck = self.fused_cond_fg_kernel[k]
 787      cb = self.fused_cond_fg_bias[k]
 788      assert ck is not None and cb is not None
 789      layer_conds.append(cond_feat @ ck + cb)
 790    return layer_conds
 791
 792  def step(
 793      self,
 794      prev_sample: float,
 795      step_layer_conds: Optional[List[np.ndarray]] = None,
 796  ) -> np.ndarray:
 797    """Computes the 256-way mu-law probability distribution for one step.
 798
 799    Args:
 800      prev_sample: Reconstructed sample `x_{i-1}` in `[-1.0, 1.0]` (use `0.0`
 801        at index `i = 0`).
 802      step_layer_conds: Optional list of `num_layers` conditioning vectors of
 803        shape `(2 * C_dil,)` for the current time step `i`.
 804
 805    Returns:
 806      1D float32 probability array `q^{(i)}` of shape `(256,)` summing to 1.
 807    """
 808    x_curr = np.float32(prev_sample)
 809    h = (
 810        self.prev_input_sample * self.in_kernel[0, 0]
 811        + x_curr * self.in_kernel[1, 0]
 812        + self.in_bias
 813    )
 814    self.prev_input_sample = x_curr
 815
 816    c_dil = self.config.dilation_channels
 817    skip_sum = np.zeros(self.config.skip_channels, dtype=np.float32)
 818
 819    for k in range(self.num_layers):
 820      ptr = self.queue_ptrs[k]
 821      h_delayed = self.layer_queues[k][ptr].copy()
 822      self.layer_queues[k][ptr] = h
 823      self.queue_ptrs[k] = (ptr + 1) % self.dilations[k]
 824
 825      fg = (
 826          h_delayed @ self.fused_fg_prev[k]
 827          + h @ self.fused_fg_curr[k]
 828          + self.fused_fg_bias[k]
 829      )
 830      if step_layer_conds is not None:
 831        fg = fg + step_layer_conds[k]
 832
 833      filt = np.tanh(fg[:c_dil])
 834      gate = 1.0 / (1.0 + np.exp(-np.clip(fg[c_dil:], -30.0, 30.0)))
 835      z = filt * gate
 836
 837      skip_sum = skip_sum + (z @ self.skip_kernels[k] + self.skip_biases[k])
 838      rk = self.res_kernels[k]
 839      rb = self.res_biases[k]
 840      if rk is not None and rb is not None:
 841        h = h + (z @ rk + rb)
 842
 843    post = np.maximum(0.0, skip_sum)
 844    post = np.maximum(0.0, post @ self.post1_kernel + self.post1_bias)
 845    logits = post @ self.logits_kernel + self.logits_bias
 846    shifted = logits - np.max(logits)
 847    exp_logits = np.exp(shifted)
 848    probs = exp_logits / np.sum(exp_logits)
 849    return probs.astype(np.float32)
 850
 851  def forward_probabilities(
 852      self,
 853      prev_samples: np.ndarray,
 854      conditioning: Optional[np.ndarray] = None,
 855  ) -> np.ndarray:
 856    """Computes `q^{(0)}, ..., q^{(T-1)}` over a sequence of past samples.
 857
 858    Args:
 859      prev_samples: 1D float array of shape `(time,)` where index `i` is
 860        `x_{i-1}` (with `prev_samples[0] == 0.0`).
 861      conditioning: Optional 2D float array of shape `(time, cond_dim)`.
 862
 863    Returns:
 864      2D float32 probability array of shape `(time, 256)`.
 865    """
 866    self.reset_state()
 867    num_steps = int(prev_samples.shape[0])
 868    probs_seq = np.zeros(
 869        (num_steps, self.config.quantization_channels), dtype=np.float32
 870    )
 871    pre_cond = None
 872    if self.use_cond:
 873      if conditioning is None:
 874        raise ValueError("Conditioning is required for parametric WaveNet.")
 875      pre_cond = self.precompute_conditioning(conditioning)
 876
 877    for i in range(num_steps):
 878      step_cond = (
 879          [pre_cond[k][i] for k in range(self.num_layers)]
 880          if pre_cond is not None
 881          else None
 882      )
 883      probs_seq[i] = self.step(float(prev_samples[i]), step_cond)
 884    return probs_seq

 886  def synthesize_parametric(
 887      self,
 888      conditioning: np.ndarray,
 889      temperature: float = 0.85,
 890      seed: int = 42,
 891  ) -> Tuple[np.ndarray, np.ndarray]:
 892    """Autoregressively synthesizes 16 kHz speech from Codec 2 conditioning.
 893
 894    At each step `i`, computes `q^{(i)}(x_i | x_{<i}, theta_{<=i})`, samples
 895    or selects quantized symbol `n_i`, reconstructs `x_i = Z(n_i)`, and feeds
 896    `x_i` into step `i + 1`.
 897
 898    Args:
 899      conditioning: Upsampled conditioning features of shape `(time, cond_dim)`.
 900      temperature: Sampling temperature (`0.0` for greedy argmax, `(0, 1]` for
 901        temperature-scaled categorical sampling).
 902      seed: Random seed for reproducible synthesis.
 903
 904    Returns:
 905      Tuple `(reconstructed_waveform, probabilities)` where:
 906        - `reconstructed_waveform`: 1D float32 array of shape `(time,)`.
 907        - `probabilities`: 2D float32 array of shape `(time, 256)`.
 908    """
 909    self.reset_state()
 910    rng = np.random.default_rng(seed)
 911    num_steps = int(conditioning.shape[0])
 912    pre_cond = self.precompute_conditioning(conditioning)
 913
 914    indices = np.zeros(num_steps, dtype=np.int32)
 915    waveform = np.zeros(num_steps, dtype=np.float32)
 916    probs_seq = np.zeros(
 917        (num_steps, self.config.quantization_channels), dtype=np.float32
 918    )
 919
 920    prev_sample = 0.0
 921    for i in range(num_steps):
 922      step_cond = [pre_cond[k][i] for k in range(self.num_layers)]
 923      probs = self.step(prev_sample, step_cond)
 924      probs_seq[i] = probs
 925
 926      # Check voicing flag (features[1] is voiced/unvoiced in Codec 2 vector)
 927      is_voiced = float(conditioning[i, 1]) > 0.5
 928      frame_energy_db = float(conditioning[i, 2]) * 60.0 - 50.0
 929      if frame_energy_db < -42.0:
 930        # Near-silence frame: suppress random sampling noise
 931        symbol = int(np.argmax(probs))
 932      elif temperature <= 1e-3:
 933        symbol = int(np.argmax(probs))
 934      else:
 935        eff_temp = temperature * (0.75 if is_voiced else 1.0)
 936        log_p = np.log(np.maximum(probs, 1e-12)) / eff_temp
 937        log_p -= np.max(log_p)
 938        scaled_p = np.exp(log_p)
 939        scaled_p /= np.sum(scaled_p)
 940        symbol = int(rng.choice(self.config.quantization_channels, p=scaled_p))
 941
 942      indices[i] = symbol
 943      sample_val = float(
 944          codec2.mulaw_dequantize(
 945              np.array([symbol], dtype=np.int32),
 946              self.config.quantization_channels - 1,
 947          )[0]
 948      )
 949      waveform[i] = sample_val
 950      prev_sample = sample_val
 951
 952    return waveform, probs_seq

 955def build_speaker_verifier(
 956    config: Optional[configs.SpeakerVerifierConfig] = None,
 957) -> tf.keras.Model:
 958  """Builds the 3-layer Projected-LSTM speaker verification network.
 959
 960  Matches the architecture used in Section 3.4 of the paper (`LstmpLayer` stack
 961  with 128 LSTM units projected to 64 dimensions and L2 normalization of the
 962  final recurrent frame state).
 963
 964  Args:
 965    config: Optional `SpeakerVerifierConfig`.
 966
 967  Returns:
 968    A `tf.keras.Model` mapping `(batch, frames, n_mels)` log-Mel filterbank
 969    features to L2-normalized speaker embeddings of shape `(batch, 64)`.
 970  """
 971  if config is None:
 972    config = configs.icassp2018_speaker_verifier_config()
 973
 974  mel_input = tf.keras.Input(
 975      shape=(None, config.n_mels),
 976      dtype=tf.float32,
 977      name="log_mel_features",
 978  )
 979  h = mel_input
 980  for idx in range(config.num_lstm_layers):
 981    h = tf.keras.layers.LSTM(
 982        units=config.lstm_units,
 983        return_sequences=True,
 984        name=f"lstm_{idx}",
 985    )(h)
 986    act = "tanh" if idx < config.num_lstm_layers - 1 else None
 987    h = tf.keras.layers.Dense(
 988        units=config.projection_dim,
 989        activation=act,
 990        name=f"lstm_proj_{idx}",
 991    )(h)
 992
 993  last_frame = tf.keras.layers.Lambda(
 994      lambda x: x[:, -1, :], name="last_frame"
 995  )(h)
 996  normed = tf.keras.layers.Lambda(
 997      lambda x: tf.math.l2_normalize(x, axis=-1),
 998      name="l2_normalized_embedding",
 999  )(last_frame)
1000  return tf.keras.Model(
1001      inputs=mel_input,
1002      outputs=normed,
1003      name="ge2e_speaker_verifier",
1004  )

1007def fit_speaker_verifier_lstmp_weights(
1008    verifier: tf.keras.Model,
1009    config: configs.SpeakerVerifierConfig,
1010    mel_sequences: Optional[np.ndarray] = None,

320def compute_subband_log_mag(waveform: np.ndarray) -> np.ndarray:
321  """Computes 32-band (0-8 kHz) RMS-normalized log-magnitude features."""
322  wav = np.asarray(waveform, dtype=np.float32).reshape(-1)
323  rms = float(np.sqrt(np.mean(wav * wav) + 1e-10))
324  w_norm = wav / rms
325  n_frames = max(1, w_norm.size // 160)
326  padded = np.pad(w_norm[:n_frames * 160], (80, 80))
327  n_fft = 320
328  win = np.hanning(n_fft).astype(np.float32)
329  out = np.zeros((n_frames, 32), dtype=np.float32)
330  for idx in range(n_frames):
331    seg = padded[idx * 160:idx * 160 + n_fft] * win
332    spec = np.abs(np.fft.rfft(seg, n=n_fft))[:160]
333    bands = np.mean(spec.reshape(32, 5) ** 2, axis=1)
334    out[idx] = np.log(bands + 1e-7)
335  return out

760  def precompute_conditioning(
761      self, conditioning: np.ndarray
762  ) -> List[np.ndarray]:
763    """Precomputes per-layer conditioning projections across all time steps.
764
765    Args:
766      conditioning: Array of shape `(time, cond_dim)`.
767
768    Returns:
769      List of `num_layers` arrays, each of shape `(time, 2 * C_dil)`.
770    """
771    cond = np.asarray(conditioning, dtype=np.float32)
772    if self.cond_conv_kernel is not None:
773      k_size = self.cond_conv_kernel.shape[0]
774      padded = np.pad(cond, ((k_size - 1, 0), (0, 0)), mode="constant")
775      out = np.zeros(
776          (cond.shape[0], self.config.cond_conv_channels), dtype=np.float32
777      )
778      for tap in range(k_size):
779        slice_t = padded[tap:tap + cond.shape[0]]
780        out += slice_t @ self.cond_conv_kernel[tap]
781      cond_feat = np.maximum(0.0, out + self.cond_conv_bias)
782    else:
783      cond_feat = cond
784
785    layer_conds: List[np.ndarray] = []
786    for k in range(self.num_layers):
787      ck = self.fused_cond_fg_kernel[k]
788      cb = self.fused_cond_fg_bias[k]
789      assert ck is not None and cb is not None
790      layer_conds.append(cond_feat @ ck + cb)
791    return layer_conds

793  def step(
794      self,
795      prev_sample: float,
796      step_layer_conds: Optional[List[np.ndarray]] = None,
797  ) -> np.ndarray:
798    """Computes the 256-way mu-law probability distribution for one step.
799
800    Args:
801      prev_sample: Reconstructed sample `x_{i-1}` in `[-1.0, 1.0]` (use `0.0`
802        at index `i = 0`).
803      step_layer_conds: Optional list of `num_layers` conditioning vectors of
804        shape `(2 * C_dil,)` for the current time step `i`.
805
806    Returns:
807      1D float32 probability array `q^{(i)}` of shape `(256,)` summing to 1.
808    """
809    x_curr = np.float32(prev_sample)
810    h = (
811        self.prev_input_sample * self.in_kernel[0, 0]
812        + x_curr * self.in_kernel[1, 0]
813        + self.in_bias
814    )
815    self.prev_input_sample = x_curr
816
817    c_dil = self.config.dilation_channels
818    skip_sum = np.zeros(self.config.skip_channels, dtype=np.float32)
819
820    for k in range(self.num_layers):
821      ptr = self.queue_ptrs[k]
822      h_delayed = self.layer_queues[k][ptr].copy()
823      self.layer_queues[k][ptr] = h
824      self.queue_ptrs[k] = (ptr + 1) % self.dilations[k]
825
826      fg = (
827          h_delayed @ self.fused_fg_prev[k]
828          + h @ self.fused_fg_curr[k]
829          + self.fused_fg_bias[k]
830      )
831      if step_layer_conds is not None:
832        fg = fg + step_layer_conds[k]
833
834      filt = np.tanh(fg[:c_dil])
835      gate = 1.0 / (1.0 + np.exp(-np.clip(fg[c_dil:], -30.0, 30.0)))
836      z = filt * gate
837
838      skip_sum = skip_sum + (z @ self.skip_kernels[k] + self.skip_biases[k])
839      rk = self.res_kernels[k]
840      rb = self.res_biases[k]
841      if rk is not None and rb is not None:
842        h = h + (z @ rk + rb)
843
844    post = np.maximum(0.0, skip_sum)
845    post = np.maximum(0.0, post @ self.post1_kernel + self.post1_bias)
846    logits = post @ self.logits_kernel + self.logits_bias
847    shifted = logits - np.max(logits)
848    exp_logits = np.exp(shifted)
849    probs = exp_logits / np.sum(exp_logits)
850    return probs.astype(np.float32)

925      probs_seq[i] = probs

927      # Check voicing flag (features[1] is voiced/unvoiced in Codec 2 vector)

930      if frame_energy_db < -42.0:
931        # Near-silence frame: suppress random sampling noise
932        symbol = int(np.argmax(probs))
933      elif temperature <= 1e-3:
934        symbol = int(np.argmax(probs))
935      else:
936        eff_temp = temperature * (0.75 if is_voiced else 1.0)
937        log_p = np.log(np.maximum(probs, 1e-12)) / eff_temp
938        log_p -= np.max(log_p)
939        scaled_p = np.exp(log_p)
940        scaled_p /= np.sum(scaled_p)
941        symbol = int(rng.choice(self.config.quantization_channels, p=scaled_p))
942
943      indices[i] = symbol
944      sample_val = float(
945          codec2.mulaw_dequantize(
946              np.array([symbol], dtype=np.int32),
947              self.config.quantization_channels - 1,
948          )[0]
949      )
950      waveform[i] = sample_val
951      prev_sample = sample_val
952
953    return waveform, probs_seq

805

807      1D float32 probability array `q^{(i)}` of shape `(256,)` summing to 1.
808    """

810    h = (
811        self.prev_input_sample * self.in_kernel[0, 0]
812        + x_curr * self.in_kernel[1, 0]
813        + self.in_bias
814    )
