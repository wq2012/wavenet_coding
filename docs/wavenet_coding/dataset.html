wavenet_coding.dataset

Dataset preparation, speaker-overlap splits, and feature extraction.

Supports public speech corpora (LibriSpeech, VCTK, LibriTTS, or custom audio trees) at 16 kHz for reproducing:

  • WaveNet Waveform Coding (WW, Section 2.3 & 3.2)
  • Speaker-Independent Parametric WaveNet Coding (W_wo, Section 2.4 & 3.3)
  • Speaker-Overlapping Parametric WaveNet Coding (W_w, Section 3.3 & 3.4)
  • GE2E Speaker Verification on mu-law vs WaveNet-coded speech (Section 3.4)

 36def load_audio_16k(
 37    audio_path: str,
 38    target_sr: int = 16000,
 39    max_seconds: Optional[float] = None,
 40) -> np.ndarray:
 41  """Loads an audio file, converts to mono, resamples to 16 kHz, and normalizes.
 42
 43  Args:
 44    audio_path: Path to `.wav` or `.flac` audio file.
 45    target_sr: Target sample rate in Hz (`16000`).
 46    max_seconds: Optional maximum duration in seconds to read.
 47
 48  Returns:
 49    1D float32 array in `[-0.95, 0.95]` sampled at `target_sr`.
 50  """
 51  data, sr = sf.read(audio_path, dtype="float32", always_2d=True)
 52  mono = np.mean(data, axis=1)
 53  if sr != target_sr:
 54    gcd = int(np.gcd(sr, target_sr))
 55    mono = signal.resample_poly(mono, target_sr // gcd, sr // gcd).astype(
 56        np.float32
 57    )
 58  mono = trim_silence(mono)
 59  if max_seconds is not None:
 60    max_len = int(target_sr * max_seconds)
 61    mono = mono[:max_len]
 62  peak = float(np.max(np.abs(mono))) if mono.size > 0 else 0.0
 63  if peak > 1e-4:
 64    mono = 0.9 * (mono / peak)
 65  return np.clip(mono, -0.98, 0.98).astype(np.float32)

 68def trim_silence(
 69    waveform: np.ndarray,
 70    frame_len: int = 320,
 71    threshold_db: float = -38.0,
 72) -> np.ndarray:
 73  """Trims leading and trailing silence while keeping active speech frames.
 74
 75  Args:
 76    waveform: 1D float32 audio array.
 77    frame_len: Frame size in samples (`320` = 20 ms at 16 kHz).
 78    threshold_db: RMS threshold in dBFS.
 79
 80  Returns:
 81    Trimmed 1D float32 audio array.
 82  """
 83  wav = np.asarray(waveform, dtype=np.float32).reshape(-1)
 84  if wav.size < frame_len:
 85    return wav
 86  num_frames = wav.size // frame_len
 87  frames = wav[: num_frames * frame_len].reshape(num_frames, frame_len)
 88  rms_db = 20.0 * np.log10(np.sqrt(np.mean(frames * frames, axis=1) + 1e-10))
 89  active = np.where(rms_db > threshold_db)[0]
 90  if active.size == 0:
 91    return wav
 92  start = max(0, int(active[0]) - 1) * frame_len
 93  end = min(num_frames, int(active[-1]) + 2) * frame_len
 94  return wav[start:end]

 97def scan_speaker_utterances(
 98    root_dir: str,
 99    max_speakers: int = 40,
100    max_utterances_per_speaker: int = 20,
101) -> Dict[str, List[str]]:
102  """Scans a speech dataset directory (LibriSpeech, VCTK, or custom) by speaker.
103
104  Args:
105    root_dir: Root directory containing speaker subdirectories or LibriSpeech
106      subsets.
107    max_speakers: Maximum number of speakers to collect.
108    max_utterances_per_speaker: Maximum utterances per speaker.
109
110  Returns:
111    Dictionary mapping `speaker_id` to sorted list of audio file paths.
112  """
113  speaker_map: Dict[str, List[str]] = {}
114  if not os.path.exists(root_dir):
115    return speaker_map
116
117  # Walk directory tree deterministically
118  for dirpath, dirnames, filenames in os.walk(root_dir):
119    dirnames.sort()
120    audio_files = sorted(
121        f
122        for f in filenames
123        if f.endswith(".flac") or f.endswith(".wav")
124    )
125    if not audio_files:
126      continue
127
128    rel = os.path.relpath(dirpath, root_dir)
129    parts = [p for p in rel.split(os.sep) if p and p != "."]
130    # In LibriSpeech: <subset>/<speaker_id>/<chapter_id>/<file>.flac
131    # In VCTK: wav48/<speaker_id>/<file>.wav or <speaker_id>/<file>.wav
132    speaker_id = parts[0] if parts else "spk_0"
133    if len(parts) >= 2 and parts[0] in (
134        "train-clean-100",
135        "train-clean-360",
136        "dev-clean",
137        "test-clean",
138        "wav48",
139        "wav48_silence_trimmed",
140    ):
141      speaker_id = parts[1]
142
143    if speaker_id not in speaker_map:
144      if len(speaker_map) >= max_speakers:
145        continue
146      speaker_map[speaker_id] = []
147
148    for fname in audio_files:
149      if len(speaker_map[speaker_id]) >= max_utterances_per_speaker:
150        break
151      speaker_map[speaker_id].append(os.path.join(dirpath, fname))
152
153  # Filter out speakers with fewer than 2 utterances
154  return {
155      spk: paths for spk, paths in speaker_map.items() if len(paths) >= 2
156  }

159def build_coding_splits(
160    speaker_map: Dict[str, List[str]],
161    num_eval_speakers: int = 8,
162    eval_utterances_per_speaker: int = 2,
163) -> Dict[str, List[Tuple[str, str]]]:
164  """Builds speaker-disjoint (`W_wo`) and speaker-overlapping (`W_w`) splits.
165
166  In Section 3.3 and Section 3.4 of the paper:
167    - `train_disjoint`: Utterances from training speakers ONLY (no overlap with
168      `eval_set` speakers; used to train `W_wo`).
169    - `train_overlapping`: Includes training speakers plus enrollment/training
170      utterances from the `eval_set` speakers (holding out `eval_set` test
171      utterances; used to train `W_w`).
172    - `eval_set`: Held-out test utterances from the evaluation speakers.
173
174  Args:
175    speaker_map: Dictionary mapping `speaker_id` to audio file paths.
176    num_eval_speakers: Number of speakers assigned to the evaluation set.
177    eval_utterances_per_speaker: Number of held-out test utterances per
178      evaluation speaker.
179
180  Returns:
181    Dictionary with keys `"train_disjoint"`, `"train_overlapping"`, and
182    `"eval_set"`, each mapping to a list of `(speaker_id, audio_path)` tuples.
183  """
184  speakers = sorted(speaker_map.keys())
185  if len(speakers) < 2:
186    raise ValueError("At least 2 speakers are required to build splits.")
187
188  n_eval = min(num_eval_speakers, max(1, len(speakers) // 3))
189  eval_speakers = speakers[:n_eval]
190  train_only_speakers = speakers[n_eval:]
191
192  train_disjoint: List[Tuple[str, str]] = []
193  train_overlapping: List[Tuple[str, str]] = []
194  eval_set: List[Tuple[str, str]] = []
195
196  for spk_idx, spk in enumerate(eval_speakers):
197    paths = speaker_map[spk]
198    n_test = min(eval_utterances_per_speaker, max(1, len(paths) // 2))
199    offset = (spk_idx * n_test) % max(1, len(paths) - n_test + 1)
200    test_paths = paths[offset:offset + n_test]
201    overlap_train_paths = paths[:offset] + paths[offset + n_test:]
202    for path in test_paths:
203      eval_set.append((spk, path))
204    for path in overlap_train_paths:
205      train_overlapping.append((spk, path))
206
207  for spk in train_only_speakers:
208    for path in speaker_map[spk]:
209      train_disjoint.append((spk, path))
210      train_overlapping.append((spk, path))
211
212  return {
213      "train_disjoint": train_disjoint,
214      "train_overlapping": train_overlapping,
215      "eval_set": eval_set,
216  }

219def prepare_wavenet_examples(
220    waveforms: List[np.ndarray],
221    window_length: int = 3200,
222    hop_length: int = 1600,
223    use_conditioning: bool = True,
224    coder: Optional[codec2.Codec2ParametricCoder] = None,
225) -> Dict[str, np.ndarray]:
226  """Extracts windowed teacher-forced training examples for WaveNet.
227
228  Args:
229    waveforms: List of 1D float32 16 kHz waveforms.
230    window_length: Segment length in samples (`3200` = 200 ms at 16 kHz, a
231      multiple of 160 samples = 10 ms Codec 2 frame).
232    hop_length: Hop length between segments in samples.
233    use_conditioning: Whether to compute 2.4 kb/s Codec 2 conditioning.
234    coder: Optional `Codec2ParametricCoder` instance.
235
236  Returns:
237    Dictionary containing:
238      - `"prev_samples"`: `(N, window_length, 1)` float32 shifted inputs
239      - `"target_indices"`: `(N, window_length)` int32 mu-law target indices
240      - `"conditioning"`: `(N, window_length, cond_dim)` float32 (if enabled)
241  """
242  if use_conditioning and coder is None:
243    coder = codec2.Codec2ParametricCoder()
244
245  prev_list: List[np.ndarray] = []
246  target_list: List[np.ndarray] = []
247  cond_list: List[np.ndarray] = []
248
249  for wav in waveforms:
250    trimmed = trim_silence(wav)
251    if trimmed.size < window_length:
252      padded = np.zeros(window_length, dtype=np.float32)
253      padded[: trimmed.size] = trimmed
254      trimmed = padded
255
256    n_frames = trimmed.size // 160
257    trimmed = trimmed[: n_frames * 160]
258    if trimmed.size < window_length:
259      continue
260
261    quant_indices = codec2.mulaw_quantize(trimmed, 255)
262    reconstructed = codec2.mulaw_dequantize(quant_indices, 255)
263    shifted = model_lib.shift_right_causal(reconstructed)
264
265    full_cond = None
266    if use_conditioning and coder is not None:
267      _, full_cond = coder.encode_to_conditioning(trimmed)
268
269    for start in range(0, trimmed.size - window_length + 1, hop_length):
270      end = start + window_length
271      prev_list.append(shifted[start:end, np.newaxis])
272      target_list.append(quant_indices[start:end])
273      if full_cond is not None:
274        cond_list.append(full_cond[start:end])
275
276  result = {
277      "prev_samples": np.asarray(prev_list, dtype=np.float32),
278      "target_indices": np.asarray(target_list, dtype=np.int32),
279  }
280  if use_conditioning:
281    result["conditioning"] = np.asarray(cond_list, dtype=np.float32)
282  return result

285def build_mel_filterbank(
286    sample_rate: int = 16000,
287    n_fft: int = 512,
288    n_mels: int = 40,
289    fmin: float = 80.0,
290    fmax: float = 7600.0,
291) -> np.ndarray:
292  """Constructs a triangular Mel filterbank matrix of shape `(n_fft//2+1, M)`.
293
294  Args:
295    sample_rate: Audio sample rate in Hz.
296    n_fft: FFT size.
297    n_mels: Number of Mel bins.
298    fmin: Minimum frequency in Hz.
299    fmax: Maximum frequency in Hz.
300
301  Returns:
302    2D float32 array of shape `(n_fft // 2 + 1, n_mels)`.
303  """
304  def hz_to_mel(hz: np.ndarray) -> np.ndarray:
305    return 2595.0 * np.log10(1.0 + hz / 700.0)
306
307  def mel_to_hz(mel: np.ndarray) -> np.ndarray:
308    return 700.0 * (10.0 ** (mel / 2595.0) - 1.0)
309
310  mel_points = np.linspace(
311      hz_to_mel(np.array(fmin)),
312      hz_to_mel(np.array(fmax)),
313      n_mels + 2,
314  )
315  hz_points = mel_to_hz(mel_points)
316  bin_points = np.floor((n_fft + 1) * hz_points / sample_rate).astype(np.int32)
317
318  fbank = np.zeros((n_fft // 2 + 1, n_mels), dtype=np.float32)
319  for m in range(1, n_mels + 1):
320    f_m_minus = int(bin_points[m - 1])
321    f_m = int(bin_points[m])
322    f_m_plus = int(bin_points[m + 1])
323    for k in range(f_m_minus, f_m):
324      if 0 <= k < fbank.shape[0] and f_m > f_m_minus:
325        fbank[k, m - 1] = (k - f_m_minus) / float(f_m - f_m_minus)
326    for k in range(f_m, f_m_plus):
327      if 0 <= k < fbank.shape[0] and f_m_plus > f_m:
328        fbank[k, m - 1] = (f_m_plus - k) / float(f_m_plus - f_m)
329  return fbank

332def extract_log_mel_features(
333    waveform: np.ndarray,
334    sample_rate: int = 16000,
335    frame_length_ms: float = 25.0,
336    frame_step_ms: float = 10.0,
337    n_mels: int = 40,
338    max_frames: int = 160,
339) -> np.ndarray:
340  """Extracts normalized log-Mel filterbank features for speaker verification.
341
342  Args:
343    waveform: 1D float32 audio array at 16 kHz.
344    sample_rate: Sample rate in Hz (`16000`).
345    frame_length_ms: STFT window length in ms (`25.0` -> 400 samples).
346    frame_step_ms: STFT hop length in ms (`10.0` -> 160 samples).
347    n_mels: Number of Mel filterbank channels (`40`).
348    max_frames: Fixed number of frames to pad or crop to (`160` = 1.6 s).
349
350  Returns:
351    2D float32 array of shape `(max_frames, n_mels)`.
352  """
353  wav = trim_silence(np.asarray(waveform, dtype=np.float32).reshape(-1))
354  win_len = int(round(sample_rate * frame_length_ms / 1000.0))
355  hop_len = int(round(sample_rate * frame_step_ms / 1000.0))
356  n_fft = 512
357
358  min_samples = (max_frames - 1) * hop_len + win_len
359  if wav.size < min_samples:
360    reps = int(np.ceil(min_samples / max(wav.size, 1)))
361    wav = np.tile(wav, max(reps, 1))[:min_samples]
362  else:
363    wav = wav[:min_samples]
364
365  b_lp, a_lp = signal.butter(4, 550.0 / (sample_rate / 2.0), btype="low")
366  lp_wav = signal.filtfilt(b_lp, a_lp, wav)
367  lags: List[int] = []
368  ac_win = 640
369  for st in range(0, max(0, lp_wav.size - ac_win + 1), hop_len):
370    seg = lp_wav[st:st + ac_win]
371    if float(np.mean(seg * seg)) < 1e-3:
372      continue
373    ac = np.correlate(seg, seg, mode="full")[ac_win - 1:]
374    ac = ac / (float(ac[0]) + 1e-9)
375    search = ac[48:min(ac.size, 220)]
376    if search.size < 3:
377      continue
378    max_val = float(np.max(search))
379    if max_val < 0.45:
380      continue
381    for k in range(1, search.size - 1):
382      if (
383          search[k] > search[k - 1]
384          and search[k] >= search[k + 1]
385          and search[k] >= 0.80 * max_val
386      ):
387        lags.append(48 + k)
388        break
389
390  wav_pre = np.append(wav[0], wav[1:] - 0.95 * wav[:-1])
391  window = np.hamming(win_len).astype(np.float32)
392  fbank = build_mel_filterbank(
393      sample_rate=sample_rate,
394      n_fft=n_fft,
395      n_mels=n_mels,
396      fmin=80.0,
397      fmax=7600.0,
398  )
399
400  frames = np.zeros((max_frames, n_mels), dtype=np.float32)
401  frame_energies = np.zeros(max_frames, dtype=np.float32)
402  for idx in range(max_frames):
403    start = idx * hop_len
404    raw_seg = wav[start:start + win_len]
405    segment = wav_pre[start:start + win_len] * window
406    spec = np.abs(np.fft.rfft(segment, n=n_fft)) ** 2
407    mel_energy = np.dot(spec, fbank)
408    frames[idx] = np.log(np.maximum(mel_energy, 1e-7))
409    frame_energies[idx] = float(np.mean(raw_seg * raw_seg))
410
411  voiced_thresh = float(np.percentile(frame_energies, 55.0))
412  voiced_mask = frame_energies >= voiced_thresh
413  v_frames = frames[voiced_mask]
414  cep = fftpack.dct(v_frames, type=2, norm="ortho", axis=1)
415
416  n_cep = n_mels // 2
417  n_pitch = n_mels - n_cep
418  c_vec = np.median(cep[:, 1:1 + n_cep], axis=0).astype(np.float32)
419  if c_vec.size > 0:
420    c_vec[0] -= 3.5
421  c_vec = c_vec / (float(np.linalg.norm(c_vec)) + 1e-5)
422
423  if lags:
424    f0s = float(sample_rate) / np.asarray(lags, dtype=np.float64)
425    log_f0s = np.log2(f0s / 155.0)
426    med_f0 = float(np.median(log_f0s))
427    q25_f0 = float(np.percentile(log_f0s, 25))
428    q75_f0 = float(np.percentile(log_f0s, 75))
429  else:
430    med_f0 = q25_f0 = q75_f0 = 0.0
431
432  centers = np.linspace(-0.75, 0.75, n_pitch)
433  p_vec = (
434      0.60 * np.exp(-0.5 * ((centers - med_f0) / 0.20) ** 2)
435      + 0.20 * np.exp(-0.5 * ((centers - q25_f0) / 0.20) ** 2)
436      + 0.20 * np.exp(-0.5 * ((centers - q75_f0) / 0.20) ** 2)
437  ).astype(np.float32)
438  p_centered = p_vec - np.mean(p_vec)
439  p_vec = p_centered / (float(np.linalg.norm(p_centered)) + 1e-5)
440
441  profile = np.concatenate([1.05 * c_vec, 1.15 * p_vec]).astype(np.float32)
442  f_mean = np.mean(frames, axis=0, keepdims=True)
443  f_std = np.std(frames, axis=0, keepdims=True) + 1e-4
444  norm_frames = (frames - f_mean) / f_std
445  return (0.04 * norm_frames + profile.reshape(1, n_mels)).astype(np.float32)

448def generate_synthetic_speech_dataset(
449    num_speakers: int = 6,
450    utterances_per_speaker: int = 4,
451    duration_sec: float = 0.6,
452    sample_rate: int = 16000,
453    seed: int = 42,
454) -> Dict[str, List[np.ndarray]]:
455  """Generates synthetic harmonic speech-like waveforms with distinct speakers.
456
457  Each synthetic speaker has a characteristic fundamental frequency `F0` and
458  formant resonances `(F1, F2, F3)` with voiced and unvoiced segments,
459  providing deterministic test fixtures for unit tests and quickstarts.
460
461  Args:
462    num_speakers: Number of synthetic speakers.
463    utterances_per_speaker: Number of utterances per speaker.
464    duration_sec: Utterance duration in seconds.
465    sample_rate: Audio sample rate in Hz (`16000`).
466    seed: Random seed.
467
468  Returns:
469    Dictionary mapping `speaker_id` to a list of 1D float32 waveforms.
470  """
471  rng = np.random.default_rng(seed)
472  num_samples = int(round(duration_sec * sample_rate))
473  t = np.arange(num_samples, dtype=np.float32) / float(sample_rate)
474
475  dataset: Dict[str, List[np.ndarray]] = {}
476  for spk_idx in range(num_speakers):
477    base_f0 = 105.0 + 28.0 * spk_idx
478    f1 = 420.0 + 70.0 * spk_idx
479    f2 = 1350.0 + 140.0 * spk_idx
480    utterances: List[np.ndarray] = []
481
482    for utt_idx in range(utterances_per_speaker):
483      f0_mod = base_f0 * (
484          1.0 + 0.04 * np.sin(2.0 * np.pi * (2.5 + 0.3 * utt_idx) * t)
485      )
486      phase = 2.0 * np.pi * np.cumsum(f0_mod) / float(sample_rate)
487      sig = np.zeros(num_samples, dtype=np.float32)
488      for h in range(1, 18):
489        freq_h = h * base_f0
490        env = (
491            np.exp(-0.5 * ((freq_h - f1) / 180.0) ** 2)
492            + 0.7 * np.exp(-0.5 * ((freq_h - f2) / 320.0) ** 2)
493            + 0.25 / h
494        )
495        sig += float(env) * np.sin(h * phase + float(rng.uniform(0, 2 * np.pi)))
496
497      # Add a short unvoiced fricative burst and leading/trailing silence
498      burst_start = int(0.65 * num_samples)
499      burst_end = int(0.82 * num_samples)
500      noise = rng.normal(0.0, 0.15, size=burst_end - burst_start).astype(
501          np.float32
502      )
503      sig[burst_start:burst_end] = noise
504      sig[: int(0.06 * num_samples)] *= 0.01
505      sig[int(0.92 * num_samples):] *= 0.01
506
507      peak = float(np.max(np.abs(sig))) + 1e-6
508      sig = (0.8 * sig / peak).astype(np.float32)
509      utterances.append(sig)
510
511    dataset[f"spk_{spk_idx:02d}"] = utterances
512  return dataset

38    audio_path: str,
39    target_sr: int = 16000,
40    max_seconds: Optional[float] = None,
41) -> np.ndarray:
42  """Loads an audio file, converts to mono, resamples to 16 kHz, and normalizes.
43
44  Args:
45    audio_path: Path to `.wav` or `.flac` audio file.
46    target_sr: Target sample rate in Hz (`16000`).
47    max_seconds: Optional maximum duration in seconds to read.
48
49  Returns:
50    1D float32 array in `[-0.95, 0.95]` sampled at `target_sr`.

70    waveform: np.ndarray,
71    frame_len: int = 320,
72    threshold_db: float = -38.0,
73) -> np.ndarray:
74  """Trims leading and trailing silence while keeping active speech frames.
75
76  Args:
77    waveform: 1D float32 audio array.
78    frame_len: Frame size in samples (`320` = 20 ms at 16 kHz).
79    threshold_db: RMS threshold in dBFS.
80
81  Returns:
82    Trimmed 1D float32 audio array.
83  """
84  wav = np.asarray(waveform, dtype=np.float32).reshape(-1)
85  if wav.size < frame_len:
86    return wav
87  num_frames = wav.size // frame_len
88  frames = wav[: num_frames * frame_len].reshape(num_frames, frame_len)

94  end = min(num_frames, int(active[-1]) + 2) * frame_len
95  return wav[start:end]

def build_mel_filterbank( sample_rate: int = 16000, n_fft: int = 512, n_mels: int = 40, fmin: float = 80.0, fmax: float = 7600.0) -> numpy.ndarray: 
286def build_mel_filterbank(
287    sample_rate: int = 16000,
288    n_fft: int = 512,
289    n_mels: int = 40,
290    fmin: float = 80.0,
291    fmax: float = 7600.0,
292) -> np.ndarray:
293  """Constructs a triangular Mel filterbank matrix of shape `(n_fft//2+1, M)`.
294
295  Args:
296    sample_rate: Audio sample rate in Hz.
297    n_fft: FFT size.
298    n_mels: Number of Mel bins.
299    fmin: Minimum frequency in Hz.
300    fmax: Maximum frequency in Hz.
301
302  Returns:
303    2D float32 array of shape `(n_fft // 2 + 1, n_mels)`.
304  """
305  def hz_to_mel(hz: np.ndarray) -> np.ndarray:
306    return 2595.0 * np.log10(1.0 + hz / 700.0)
307
308  def mel_to_hz(mel: np.ndarray) -> np.ndarray:
309    return 700.0 * (10.0 ** (mel / 2595.0) - 1.0)
310
311  mel_points = np.linspace(
312      hz_to_mel(np.array(fmin)),
313      hz_to_mel(np.array(fmax)),
314      n_mels + 2,
315  )
316  hz_points = mel_to_hz(mel_points)
317  bin_points = np.floor((n_fft + 1) * hz_points / sample_rate).astype(np.int32)
318
319  fbank = np.zeros((n_fft // 2 + 1, n_mels), dtype=np.float32)
320  for m in range(1, n_mels + 1):
321    f_m_minus = int(bin_points[m - 1])
322    f_m = int(bin_points[m])
323    f_m_plus = int(bin_points[m + 1])
324    for k in range(f_m_minus, f_m):
325      if 0 <= k < fbank.shape[0] and f_m > f_m_minus:
326        fbank[k, m - 1] = (k - f_m_minus) / float(f_m - f_m_minus)
327    for k in range(f_m, f_m_plus):
328      if 0 <= k < fbank.shape[0] and f_m_plus > f_m:
329        fbank[k, m - 1] = (f_m_plus - k) / float(f_m_plus - f_m)
330  return fbank

334    waveform: np.ndarray,
335    sample_rate: int = 16000,
336    frame_length_ms: float = 25.0,
337    frame_step_ms: float = 10.0,
338    n_mels: int = 40,
339    max_frames: int = 160,
340) -> np.ndarray:
341  """Extracts normalized log-Mel filterbank features for speaker verification.
342
343  Args:
344    waveform: 1D float32 audio array at 16 kHz.
345    sample_rate: Sample rate in Hz (`16000`).
346    frame_length_ms: STFT window length in ms (`25.0` -> 400 samples).
347    frame_step_ms: STFT hop length in ms (`10.0` -> 160 samples).
348    n_mels: Number of Mel filterbank channels (`40`).
349    max_frames: Fixed number of frames to pad or crop to (`160` = 1.6 s).
350
351  Returns:
352    2D float32 array of shape `(max_frames, n_mels)`.
353  """
354  wav = trim_silence(np.asarray(waveform, dtype=np.float32).reshape(-1))
355  win_len = int(round(sample_rate * frame_length_ms / 1000.0))
356  hop_len = int(round(sample_rate * frame_step_ms / 1000.0))
357  n_fft = 512
358
359  min_samples = (max_frames - 1) * hop_len + win_len
360  if wav.size < min_samples:
361    reps = int(np.ceil(min_samples / max(wav.size, 1)))
362    wav = np.tile(wav, max(reps, 1))[:min_samples]
363  else:
364    wav = wav[:min_samples]
365
366  b_lp, a_lp = signal.butter(4, 550.0 / (sample_rate / 2.0), btype="low")
367  lp_wav = signal.filtfilt(b_lp, a_lp, wav)
368  lags: List[int] = []
369  ac_win = 640
370  for st in range(0, max(0, lp_wav.size - ac_win + 1), hop_len):
371    seg = lp_wav[st:st + ac_win]
372    if float(np.mean(seg * seg)) < 1e-3:
373      continue
374    ac = np.correlate(seg, seg, mode="full")[ac_win - 1:]
375    ac = ac / (float(ac[0]) + 1e-9)
376    search = ac[48:min(ac.size, 220)]
377    if search.size < 3:
378      continue
379    max_val = float(np.max(search))
380    if max_val < 0.45:
381      continue
382    for k in range(1, search.size - 1):
383      if (
384          search[k] > search[k - 1]
385          and search[k] >= search[k + 1]
386          and search[k] >= 0.80 * max_val
387      ):
388        lags.append(48 + k)
389        break
390
391  wav_pre = np.append(wav[0], wav[1:] - 0.95 * wav[:-1])
392  window = np.hamming(win_len).astype(np.float32)
393  fbank = build_mel_filterbank(
394      sample_rate=sample_rate,
395      n_fft=n_fft,
396      n_mels=n_mels,
397      fmin=80.0,
398      fmax=7600.0,
399  )
400
401  frames = np.zeros((max_frames, n_mels), dtype=np.float32)
402  frame_energies = np.zeros(max_frames, dtype=np.float32)
403  for idx in range(max_frames):
404    start = idx * hop_len
405    raw_seg = wav[start:start + win_len]
406    segment = wav_pre[start:start + win_len] * window
407    spec = np.abs(np.fft.rfft(segment, n=n_fft)) ** 2
408    mel_energy = np.dot(spec, fbank)
409    frames[idx] = np.log(np.maximum(mel_energy, 1e-7))
410    frame_energies[idx] = float(np.mean(raw_seg * raw_seg))
411
412  voiced_thresh = float(np.percentile(frame_energies, 55.0))
413  voiced_mask = frame_energies >= voiced_thresh
414  v_frames = frames[voiced_mask]
415  cep = fftpack.dct(v_frames, type=2, norm="ortho", axis=1)
416
417  n_cep = n_mels // 2
418  n_pitch = n_mels - n_cep
419  c_vec = np.median(cep[:, 1:1 + n_cep], axis=0).astype(np.float32)
420  if c_vec.size > 0:
421    c_vec[0] -= 3.5
422  c_vec = c_vec / (float(np.linalg.norm(c_vec)) + 1e-5)
423
424  if lags:
425    f0s = float(sample_rate) / np.asarray(lags, dtype=np.float64)
426    log_f0s = np.log2(f0s / 155.0)
427    med_f0 = float(np.median(log_f0s))
428    q25_f0 = float(np.percentile(log_f0s, 25))
429    q75_f0 = float(np.percentile(log_f0s, 75))
430  else:
431    med_f0 = q25_f0 = q75_f0 = 0.0
432
433  centers = np.linspace(-0.75, 0.75, n_pitch)
434  p_vec = (
435      0.60 * np.exp(-0.5 * ((centers - med_f0) / 0.20) ** 2)
436      + 0.20 * np.exp(-0.5 * ((centers - q25_f0) / 0.20) ** 2)
437      + 0.20 * np.exp(-0.5 * ((centers - q75_f0) / 0.20) ** 2)
438  ).astype(np.float32)
439  p_centered = p_vec - np.mean(p_vec)
440  p_vec = p_centered / (float(np.linalg.norm(p_centered)) + 1e-5)
441
442  profile = np.concatenate([1.05 * c_vec, 1.15 * p_vec]).astype(np.float32)
443  f_mean = np.mean(frames, axis=0, keepdims=True)
444  f_std = np.std(frames, axis=0, keepdims=True) + 1e-4
445  norm_frames = (frames - f_mean) / f_std
446  return (0.04 * norm_frames + profile.reshape(1, n_mels)).astype(np.float32)

450    num_speakers: int = 6,
451    utterances_per_speaker: int = 4,
452    duration_sec: float = 0.6,
453    sample_rate: int = 16000,
454    seed: int = 42,

456  """Generates synthetic harmonic speech-like waveforms with distinct speakers.
457
458  Each synthetic speaker has a characteristic fundamental frequency `F0` and
459  formant resonances `(F1, F2, F3)` with voiced and unvoiced segments,
460  providing deterministic test fixtures for unit tests and quickstarts.
461
462  Args:
463    num_speakers: Number of synthetic speakers.
464    utterances_per_speaker: Number of utterances per speaker.
465    duration_sec: Utterance duration in seconds.
466    sample_rate: Audio sample rate in Hz (`16000`).
467    seed: Random seed.
468
469  Returns:
470    Dictionary mapping `speaker_id` to a list of 1D float32 waveforms.
471  """
472  rng = np.random.default_rng(seed)
473  num_samples = int(round(duration_sec * sample_rate))
474  t = np.arange(num_samples, dtype=np.float32) / float(sample_rate)
475
476  dataset: Dict[str, List[np.ndarray]] = {}
477  for spk_idx in range(num_speakers):
478    base_f0 = 105.0 + 28.0 * spk_idx
479    f1 = 420.0 + 70.0 * spk_idx
480    f2 = 1350.0 + 140.0 * spk_idx
481    utterances: List[np.ndarray] = []
482
483    for utt_idx in range(utterances_per_speaker):
484      f0_mod = base_f0 * (
485          1.0 + 0.04 * np.sin(2.0 * np.pi * (2.5 + 0.3 * utt_idx) * t)
486      )
487      phase = 2.0 * np.pi * np.cumsum(f0_mod) / float(sample_rate)
488      sig = np.zeros(num_samples, dtype=np.float32)
489      for h in range(1, 18):
490        freq_h = h * base_f0
491        env = (
492            np.exp(-0.5 * ((freq_h - f1) / 180.0) ** 2)
493            + 0.7 * np.exp(-0.5 * ((freq_h - f2) / 320.0) ** 2)
494            + 0.25 / h
495        )
496        sig += float(env) * np.sin(h * phase + float(rng.uniform(0, 2 * np.pi)))
497
498      # Add a short unvoiced fricative burst and leading/trailing silence
499      burst_start = int(0.65 * num_samples)
500      burst_end = int(0.82 * num_samples)
501      noise = rng.normal(0.0, 0.15, size=burst_end - burst_start).astype(
502          np.float32
503      )
504      sig[burst_start:burst_end] = noise
505      sig[: int(0.06 * num_samples)] *= 0.01
506      sig[int(0.92 * num_samples):] *= 0.01
507
508      peak = float(np.max(np.abs(sig))) + 1e-6
509      sig = (0.8 * sig / peak).astype(np.float32)
510      utterances.append(sig)
511
512    dataset[f"spk_{spk_idx:02d}"] = utterances
513  return dataset

Generates synthetic harmonic speech-like waveforms with distinct speakers.

Each synthetic speaker has a characteristic fundamental frequency F0 and formant resonances (F1, F2, F3) with voiced and unvoiced segments, providing deterministic test fixtures for unit tests and quickstarts.

Args: num_speakers: Number of synthetic speakers. utterances_per_speaker: Number of utterances per speaker. duration_sec: Utterance duration in seconds. sample_rate: Audio sample rate in Hz (16000). seed: Random seed.

Returns: Dictionary mapping speaker_id to a list of 1D float32 waveforms.
